cmake_minimum_required(VERSION 3.16)
project(LocalAI VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network)
find_package(Qt6 QUIET OPTIONAL_COMPONENTS Pdf)
qt_add_executable(LocalAI main.cpp)
target_link_libraries(LocalAI PRIVATE Qt6::Widgets Qt6::Network)
if(TARGET Qt6::Pdf)
    message(STATUS "Qt PDF gefunden - PDF-Anhaenge aktiv")
    target_link_libraries(LocalAI PRIVATE Qt6::Pdf)
    target_compile_definitions(LocalAI PRIVATE HAVE_QTPDF)
else()
    message(STATUS "Qt PDF nicht gefunden - PDF-Anhaenge deaktiviert")
endif()
set_target_properties(LocalAI PROPERTIES WIN32_EXECUTABLE ON)
